% Kabyle (kab-Latn) postprocessor
% Implements dynamic phonological rules for spirantization, vowel backing, and nasal assimilation.
% Uses uppercase placeholders for geminates to protect them from spirantization.

::vowels:: = a|æ|ɑ|i|ɪ|u|ʊ|ə|e|o|ɛ|ɔ
::emphatics:: = ṣ|ḍ|ṭ|ẓ|ṛ|sˤ|dˤ|tˤ|zˤ|rˤ|Ṭ|Ḍ
::uvulars:: = q|qː|x|xː|χ|χː|ɣ|ɣː|ʁ|ʁː
::dorsals:: = k|K|g|G|q|qː|x|xː|χ|χː|ɣ|ɣː|ʁ|ʁː
::palatals:: = j|ʒ|ʃ|d͡ʒ|t͡ʃ|d͡ʒː|t͡ʃː

% 1. Default Vowel Realization
% Epitran map.csv maps 'a' -> 'a'. In Kabyle, default 'a' is [æ].
a -> æ / _

% 2. Vowel Backing (Emphatic/Uvular Spread)
% /æ/ backs to [ɑ] near emphatic and uvular consonants
æ -> ɑ / _ (::emphatics::|::uvulars::)
æ -> ɑ / (::emphatics::|::uvulars::) _

% 3. Nasal Place Assimilation
% /n/ assimilates to [ŋ] before dorsals and palatals
n -> ŋ / _ (::dorsals::|::palatals::)
% /n/ assimilates to [m] before labials
n -> m / _ (b|B|p|f|m)

% 4. Spirantization (Positive Constraints)
% We ONLY apply spirantization post-vocalically, word-initially (#),
% or after specific sonorants. This inherently BLOCKS it
% after homorganic nasals (mb, nt, nd) and other consonants (lt).

% Bilabial /b/ -> [β]
b -> β / # _
b -> β / ::vowels:: _

% Dental /d/ -> [ð]
% NO word-initial rule for d (matches o2i's KAB_BLOCK_SPIRANT_D_WORD_INITIAL)
d -> ð / ::vowels:: _
d -> ð / r _
d -> ð / rˤ _

% Emphatic Dental /ḍ/ -> [ðˤ]
% NO word-initial rule for ḍ
Ḍ -> ðˤ / ::vowels:: _
Ḍ -> ðˤ / n _
Ḍ -> ðˤ / m _
Ḍ -> ðˤ / l _

% Dental /t/ -> [θ]
t -> θ / # _
t -> θ / ::vowels:: _

% Velar /k/ -> [ç] (Spirantizes post-vocalically, initially, and after /r/)
k -> ç / # _
k -> ç / ::vowels:: _
k -> ç / r _
k -> ç / rˤ _

% Velar /g/ -> [ʝ] (Spirantizes post-vocalically, initially, and after /m/)
% Note: Blocked after /r/ (argaz -> [arɡaz])
ɡ -> ʝ / # _
ɡ -> ʝ / ::vowels:: _
ɡ -> ʝ / m _

% 5. Restore Placeholders (Lowercase them back)
% This must happen AFTER all spirantization rules
Bː -> bː / _
Dː -> dː / _
Tː -> tː / _
Kː -> kː / _
Gː -> ɡː / _
Ṭ -> tˤ / _
Ṭː -> tˤː / _
Ḍ -> dˤ / _
Ḍː -> dˤː / _
